cmake_minimum_required(VERSION 3.22)
project(LemonVocalEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
file(GLOB WORLD_SOURCES CONFIGURE_DEPENDS vendor/world/src/*.cpp)
add_library(LemonVocalEngine SHARED bridge.cpp ${WORLD_SOURCES})
target_include_directories(LemonVocalEngine PRIVATE vendor/world/src)
set_target_properties(LemonVocalEngine PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
